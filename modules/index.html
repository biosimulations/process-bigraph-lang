    Modules¶

    The public Python API of v2: text in, checked documents out; documents in, text out. All of it runs in the DSL CLI (cli-native, TypeScript): parsing, checking, compiling and decompiling. There is no Python model of a composite, so a Python tool that builds composites targets the DSL text, where the result is checked, reviewed and documented (docs/v2/design.md, section 5.3).

    • compile_text / compile_file: .pblang text -> parse -> check -> a process-bigraph {"schema", "state"} document or a CompositeSpec mapping. An error-level finding stops the compile (CompileError carries every finding); warnings and infos come back on the result.
    • decompile: a document or a CompositeSpec mapping -> canonical .pblang text plus sidecar assets.

    Documents cross the CLI boundary as JSON in bigraph_schema.json_codec's encoding (numpy arrays tagged), with tuples tagged as {"__tuple__": true, "data": [...]} so that they survive too.

    def compile_file(
    path: str | os.PathLike[str],
    *,
    registry: InterfaceRegistry | None = None,
    manifests: list[Path] | None = None,
    to: Target = "document",
    strict: bool = False,
    force: bool = False,
    addresses: AddressForm = "import",
) -> Compiled:
    """
    Compile a `.pblang` file; `file(...)` references resolve next to it. `force` compiles despite the checker's
    errors (parse errors still stop it): the escape for documents that rely on defaults the checker flags.
    """
    source = Path(path).resolve()
    with tempfile.TemporaryDirectory() as tmp:
        manifest_paths = [*(manifests or []), *registry_manifests(registry, Path(tmp))]
        diagnostics = check_file(source, manifest_paths)
        fatal = set() if force else {"error", "warning"} if strict else {"error"}
        if any(d.severity in fatal for d in diagnostics) or any(d.code == "P001" for d in diagnostics):
            raise CompileError(diagnostics)
        out = Path(tmp) / ("compiled.yaml" if to == "spec" else "compiled.json")
        args = ["compile", str(source), "--to", to, "--out", str(out), "--addresses", addresses]
        args += ["--force"] if force else []
        if manifest_paths:
            args += ["--interfaces", *[str(Path(m).resolve()) for m in manifest_paths]]
        run_cli(*args)
        text = out.read_text(encoding="utf-8")
    if to == "spec":
        import yaml

        return Compiled(yaml.safe_load(text), diagnostics)
    return Compiled(loads(text), diagnostics)

    def compile_text(
    text: str,
    *,
    registry: InterfaceRegistry | None = None,
    manifests: list[Path] | None = None,
    to: Target = "document",
    strict: bool = False,
    force: bool = False,
    assets: dict[str, bytes] | None = None,
    addresses: AddressForm = "import",
) -> Compiled:
    """Compile v2 text. `assets` holds the sidecar files `file(...)` references name, by path."""
    with tempfile.TemporaryDirectory() as tmp:
        source = Path(tmp) / "document.pblang"
        source.write_text(text, encoding="utf-8")
        for name, content in (assets or {}).items():
            target = Path(tmp) / name
            target.parent.mkdir(parents=True, exist_ok=True)
            target.write_bytes(content)
        return compile_file(
            source, registry=registry, manifests=manifests, to=to, strict=strict, force=force, addresses=addresses
        )

    def decompile(
    mapping: dict[str, Any],
    *,
    name: str = "document",
    registry: InterfaceRegistry | None = None,
    manifests: list[Path] | None = None,
) -> Printed:
    """Canonical `.pblang` text (and sidecar assets) for a document or a CompositeSpec mapping (told apart by keys)."""
    with tempfile.TemporaryDirectory() as tmp:
        source = Path(tmp) / f"{name}.json"
        source.write_text(dumps(mapping), encoding="utf-8")
        out = Path(tmp) / f"{name}.pblang"
        args = ["decompile", str(source), "--name", name, "--out", str(out)]
        manifest_paths = [*(manifests or []), *registry_manifests(registry, Path(tmp))]
        if manifest_paths:
            args += ["--interfaces", *[str(Path(m).resolve()) for m in manifest_paths]]
        run_cli(*args)
        printed = Printed(out.read_text(encoding="utf-8"))
        assets_dir = Path(tmp) / f"{name}.assets"
        if assets_dir.is_dir():
            for asset in sorted(assets_dir.iterdir()):
                printed.assets[f"{name}.assets/{asset.name}"] = asset.read_bytes()
        return printed

    describe_module imports a Python module once (the publish-time step; authoring never imports) and turns every step/process class it exposes into an InterfaceDecl: the address in portable local:!module.Class form, the config_schema, and the ports obtained by calling inputs()/outputs() with the class standing in for self (ports that depend on an instance are marked dynamic_ports). Types stay in bigraph-schema's own form. The InterfaceRegistry holds declarations and reads and writes the JSON interface manifest (pbg-interfaces/1, a format shared with other tools; the earlier pblang-manifest/0 is still read), which the TypeScript checker, printer and binder consume; render_interfaces has the CLI print the .pblang header file.

    @dataclass
class InterfaceRegistry:
    """Declarations by address: the manifest the CLI reads, and what the normalizer resolves edges in."""

    decls: list[InterfaceDecl] = field(default_factory=list)

    @classmethod
    def from_modules(cls, *module_names: str) -> InterfaceRegistry:
        registry = cls()
        for module_name in module_names:
            registry.decls.extend(describe_module(module_name))
        return registry

    @classmethod
    def from_manifest(cls, manifest: dict[str, Any]) -> InterfaceRegistry:
        version = manifest.get("schema_version")
        if version not in (None, MANIFEST_VERSION, LEGACY_MANIFEST_VERSION):
            raise ValueError(
                f"unsupported manifest schema_version {version!r}: this reader understands {MANIFEST_VERSION}"
            )
        return cls([_decl_from_manifest(entry) for entry in manifest.get("components", [])])

    def to_manifest(self) -> dict[str, Any]:
        return {
            "$schema": SCHEMA_URL,
            "schema_version": MANIFEST_VERSION,
            "components": [_decl_to_manifest(d) for d in self.decls],
        }

    def lookup(self, address: str) -> InterfaceDecl | None:
        """By exact address; else by unique short name (`local:Name`, or a class re-exported from another module)."""
        wanted = address if ":" in address else f"local:{address}"
        if not wanted.startswith("local:"):
            return None
        exact = [d for d in self.decls if d.address == wanted]
        if exact:
            return exact[0]
        by_name = [d for d in self.decls if d.name == short_name(wanted)]
        return by_name[0] if len(by_name) == 1 else None

    def write_manifest(self, path: Path) -> Path:
        path.write_text(json.dumps(self.to_manifest(), indent=2, default=str) + "\n", encoding="utf-8")
        return path

    def describe_class(cls: type, module_name: str | None = None, name: str | None = None) -> InterfaceDecl:
    """The declaration of one step/process class."""
    module_name = module_name or cls.__module__

        outputs=outputs,
        doc=doc_of(cls),

    def describe_module(module_name: str) -> list[InterfaceDecl]:
    """Declarations for the step/process classes a module defines or re-exports (classes of its own package only)."""
    module = importlib.import_module(module_name)
    is_package = hasattr(module, "__path__")
    decls: list[InterfaceDecl] = []
    for name, cls in inspect.getmembers(module, inspect.isclass):
        if name.startswith("_") or not is_edge_class(cls) or not _belongs_to(cls, module_name, is_package):
            continue
        decls.append(describe_class(cls, module_name, name))
    return decls

    describe-model for SBML: a model file's interface as data (see docs/v2/design.md, section 5.6).

    describe_sbml reads an SBML file with the standard library's XML parser (no libsbml, no process code) and returns a ModelHeader: the model's time unit, compartments, species (as concentrations unless the species has only substance units) and parameters, each with its value and its units composed from the model's unit definitions in a pint-readable spelling, plus the counts a solver cares about (reactions, events, rules) and the SBML level, version and packages. to_dict is the JSON form (pblang-model/0), which the CLI renders as the .pblang model header (print_model_header).

    def describe_sbml(path: str | Path) -> ModelHeader:
    """The model header of an SBML file."""
    path = Path(path)
    root = ET.parse(path).getroot()
    model = root.find("{*}model")
    if model is None:
        raise ValueError(f"{path}: no <model> element")
    header = ModelHeader(name=_identifier(path.stem), source=path.name, sbml_id=model.get("id"))
    header.level, header.version = _int(root.get("level")), _int(root.get("version"))
    header.packages = sorted({_package(k) for k in root.attrib if k.endswith(":required")} - {""})
    unit_defs = {d.get("id", ""): _unit_text(d) for d in model.findall("{*}listOfUnitDefinitions/{*}unitDefinition")}

    def units(reference: str | None, default_key: str) -> str | None:
        if reference:
            return unit_defs.get(reference) or _KIND_SYMBOLS.get(reference, reference)
        return unit_defs.get(model.get(f"{default_key}Units") or default_key) or _SBML_DEFAULTS.get(default_key)

    header.time_units = units(None, "time")
    for c in model.findall("{*}listOfCompartments/{*}compartment"):
        dims = _int(c.get("spatialDimensions")) if c.get("spatialDimensions") else 3
        default = {3: "volume", 2: "area", 1: "length"}.get(dims or 3, "volume")
        header.compartments[c.get("id", "")] = Quantity(_float_or_none(c.get("size")), units(c.get("units"), default))
    for sp in model.findall("{*}listOfSpecies/{*}species"):
        name = sp.get("id", "")
        compartment = sp.get("compartment")
        substance = units(sp.get("substanceUnits"), "substance")
        comp_units = (
            header.compartments[compartment].units if compartment in header.compartments else units(None, "volume")
        )
        only_substance = sp.get("hasOnlySubstanceUnits") == "true"
        if only_substance:
            initial, kind, unit_text = _float_or_none(sp.get("initialAmount")), "amount", substance
        elif sp.get("initialConcentration") is not None:
            initial, kind, unit_text = (
                _float_or_none(sp.get("initialConcentration")),
                "concentration",
                _ratio(substance, comp_units),
            )
        else:
            amount = _float_or_none(sp.get("initialAmount"))
            size = header.compartments[compartment].value if compartment in header.compartments else None
            initial = amount / size if amount is not None and size else amount
            kind, unit_text = "concentration", _ratio(substance, comp_units)
            if amount is not None and size:
                header.notes.append(
                    f"species {name}: initial amount {amount:g} converted to concentration by compartment size {size:g}"
                )
        header.species[name] = SpeciesDecl(initial, unit_text, kind, compartment, constant=sp.get("constant") == "true")
    for pr in model.findall("{*}listOfParameters/{*}parameter"):
        header.parameters[pr.get("id", "")] = Quantity(
            _float_or_none(pr.get("value")),
            units(pr.get("units"), "") if pr.get("units") else None,
            pr.get("constant") != "false",
        )
    header.counts = {
        "reactions": len(model.findall("{*}listOfReactions/{*}reaction")),
        "events": len(model.findall("{*}listOfEvents/{*}event")),
        "rules": len(model.findall("{*}listOfRules/*")),
        "function_definitions": len(model.findall("{*}listOfFunctionDefinitions/{*}functionDefinition")),
    }
    return header

    Equivalence of composite documents modulo representation.

    Two documents are equivalent when they realize the same composite. normalize_document rewrites a document into one canonical shape: types are pushed through bigraph-schema (core.access + core.render(defaults=True)), addresses become protocol:data strings, wires become lists, tuples and arrays become lists, runtime keys (instance, _contract, the root global_time) are dropped, and the edge entries of the schema half are folded into the state half. equivalent compares two normalized documents; diff_documents explains a difference.

    def deep_equal(a: Any, b: Any, *, strict: bool = False, rel_tol: float = 1e-9) -> bool:
    """
    Structural equality that tolerates numpy arrays and float noise.

    With `strict=False` (the normalizer's mode) tuples equal lists and ints equal floats; with `strict=True`
    container types and numeric types must match exactly (the v1 parity tests use this).
    """
    if isinstance(a, np.ndarray) or isinstance(b, np.ndarray):
        if strict and not (isinstance(a, np.ndarray) and isinstance(b, np.ndarray)):
            return False
        return deep_equal(_to_list(a), _to_list(b), strict=False, rel_tol=rel_tol)
    if isinstance(a, dict) and isinstance(b, dict):
        return set(a) == set(b) and all(deep_equal(a[k], b[k], strict=strict, rel_tol=rel_tol) for k in a)
    if isinstance(a, (list, tuple)) and isinstance(b, (list, tuple)):
        if strict and type(a) is not type(b):
            return False
        return len(a) == len(b) and all(deep_equal(x, y, strict=strict, rel_tol=rel_tol) for x, y in zip(a, b))
    if isinstance(a, bool) or isinstance(b, bool):
        return a is b or (isinstance(a, bool) and isinstance(b, bool) and a == b)
    if _is_number(a) and _is_number(b):
        if strict and (isinstance(a, float) != isinstance(b, float)):
            return False
        return math.isclose(float(a), float(b), rel_tol=rel_tol, abs_tol=0.0)
    return bool(a == b)

    def diff_documents(
    a: dict[str, Any], b: dict[str, Any], core: Any, limit: int = 200, registry: InterfaceRegistry | None = None
) -> str:
    """A readable account of how two documents differ after normalization (empty when equivalent)."""
    left, right = normalize_document(a, core, registry), normalize_document(b, core, registry)
    flat_left, flat_right = _flatten(left), _flatten(right)
    lines: list[str] = []
    for key in sorted(set(flat_left) | set(flat_right)):
        if key not in flat_right:
            lines.append(f"- {key}: {_show(flat_left[key])}")
        elif key not in flat_left:
            lines.append(f"+ {key}: {_show(flat_right[key])}")
        elif not deep_equal(flat_left[key], flat_right[key]):
            lines.append(f"~ {key}: {_show(flat_left[key])} -> {_show(flat_right[key])}")
    if len(lines) > limit:
        lines = [*lines[:limit], f"... {len(lines) - limit} more"]
    if not lines and not deep_equal(left, right):
        dump_left = json.dumps(left, indent=2, sort_keys=True, default=str).splitlines()
        dump_right = json.dumps(right, indent=2, sort_keys=True, default=str).splitlines()
        lines = list(difflib.unified_diff(dump_left, dump_right, "a", "b", lineterm=""))
    return "\n".join(lines)

    def normalize_document(doc: dict[str, Any], core: Any, registry: InterfaceRegistry | None = None) -> dict[str, Any]:
    """
    The canonical shape of a document (see the module docstring). `core` is a bigraph-schema core. With a
    `registry`, an edge's `_inputs`/`_outputs` that merely repeat its interface's face are dropped (process-bigraph
    merges them with `interface()`, so they carry no information).
    """
    table: _EdgeTable = {}
    raw_schema = doc.get("schema") or {}
    schema = _norm_schema(raw_schema, core, (), table)
    state = _norm_state(doc.get("state") or {}, raw_schema, core, (), table, registry)
    options = {k: _canon(doc[k], core) for k in COMPOSITE_OPTION_KEYS if k in doc}
    return {"schema": schema if isinstance(schema, dict) else {}, "state": state, "options": options}
